# Demonstration B — API Integration & Troubleshooting

## Problem

A small team needs a repeatable export of public GitHub issues for triage. A reliable integration must handle pagination, optional authentication, invalid payloads, transient failures, and CSV handoff.

## Run the live public-API demo

No token is required for a small public request:

```powershell
python github_issues_exporter.py psf requests -o output/issues.csv --per-page 5 --max-pages 2
```

For higher GitHub rate limits, set an authorized token in `GITHUB_TOKEN`. The token is read from the environment and is never written to output.

## Evidence

```powershell
python -m unittest discover -s tests -v
```

The tests simulate two API pages, confirm pull requests are excluded, verify CSV output, and exercise retry/error behavior without transmitting credentials.

## Business value

The integration produces a clean, auditable handoff file and fails clearly when the API response is incomplete instead of silently exporting bad data.

github_issues_exporter.py

from __future__ import annotations

import argparse
import csv
import json
import os
import time
from dataclasses import dataclass
from pathlib import Path
from typing import Callable
from urllib.error import HTTPError, URLError
from urllib.parse import urlencode
from urllib.request import Request, urlopen


class IntegrationError(RuntimeError):
    pass


@dataclass(frozen=True)
class ApiPage:
    items: list[dict]
    next_url: str | None


def parse_next_link(raw_link: str | None) -> str | None:
    if not raw_link:
        return None
    for section in raw_link.split(","):
        parts = [part.strip() for part in section.split(";")]
        if len(parts) >= 2 and 'rel="next"' in parts[1:]:
            return parts[0].strip("<>")
    return None


class GitHubIssuesClient:
    def __init__(
        self,
        token: str | None = None,
        *,
        base_url: str = "https://api.github.com",
        timeout: int = 15,
        max_retries: int = 2,
        opener: Callable = urlopen,
        sleeper: Callable[[float], None] = time.sleep,
    ) -> None:
        self.token = token
        self.base_url = base_url.rstrip("/")
        self.timeout = timeout
        self.max_retries = max_retries
        self.opener = opener
        self.sleeper = sleeper

    def _request_page(self, url: str) -> ApiPage:
        headers = {
            "Accept": "application/vnd.github+json",
            "User-Agent": "workflow-repair-portfolio-demo",
            "X-GitHub-Api-Version": "2022-11-28",
        }
        if self.token:
            headers["Authorization"] = f"Bearer {self.token}"
        request = Request(url, headers=headers)

        for attempt in range(self.max_retries + 1):
            try:
                with self.opener(request, timeout=self.timeout) as response:
                    payload = json.loads(response.read().decode("utf-8"))
                    if not isinstance(payload, list):
                        raise IntegrationError("GitHub returned a non-list payload")
                    return ApiPage(payload, parse_next_link(response.headers.get("Link")))
            except (HTTPError, URLError, TimeoutError, json.JSONDecodeError) as exc:
                if attempt >= self.max_retries:
                    raise IntegrationError(f"GitHub request failed after {attempt + 1} attempts: {exc}") from exc
                self.sleeper(0.25 * (2**attempt))
        raise IntegrationError("unreachable retry state")

    def fetch_issues(
        self,
        owner: str,
        repo: str,
        *,
        state: str = "open",
        per_page: int = 30,
        max_pages: int = 3,
    ) -> list[dict[str, str | int]]:
        if state not in {"open", "closed", "all"}:
            raise ValueError("state must be open, closed, or all")
        params = urlencode({"state": state, "per_page": per_page, "page": 1})
        next_url: str | None = f"{self.base_url}/repos/{owner}/{repo}/issues?{params}"
        exported: list[dict[str, str | int]] = []

        for _ in range(max_pages):
            if not next_url:
                break
            page = self._request_page(next_url)
            for item in page.items:
                if "pull_request" in item:
                    continue
                missing = [key for key in ("id", "number", "title", "state", "html_url") if key not in item]
                if missing:
                    raise IntegrationError(f"Issue payload missing fields: {', '.join(missing)}")
                exported.append(
                    {
                        "id": item["id"],
                        "number": item["number"],
                        "title": " ".join(str(item["title"]).split()),
                        "state": item["state"],
                        "url": item["html_url"],
                        "updated_at": item.get("updated_at", ""),
                    }
                )
            next_url = page.next_url
        return exported


def export_csv(issues: list[dict[str, str | int]], output_path: Path) -> None:
    output_path.parent.mkdir(parents=True, exist_ok=True)
    fields = ["id", "number", "title", "state", "url", "updated_at"]
    with output_path.open("w", newline="", encoding="utf-8-sig") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields)
        writer.writeheader()
        writer.writerows(issues)


def main() -> None:
    parser = argparse.ArgumentParser(description="Export public GitHub issues to CSV.")
    parser.add_argument("owner")
    parser.add_argument("repo")
    parser.add_argument("-o", "--output", type=Path, default=Path("output/issues.csv"))
    parser.add_argument("--state", choices=("open", "closed", "all"), default="open")
    parser.add_argument("--per-page", type=int, default=30)
    parser.add_argument("--max-pages", type=int, default=3)
    args = parser.parse_args()

    client = GitHubIssuesClient(token=os.environ.get("GITHUB_TOKEN"))
    issues = client.fetch_issues(
        args.owner,
        args.repo,
        state=args.state,
        per_page=args.per_page,
        max_pages=args.max_pages,
    )
    export_csv(issues, args.output)
    print(f"Issues exported: {len(issues)}")
    print(f"Output: {args.output.resolve()}")


if __name__ == "__main__":
    main()

tests/test_github_issues_exporter.py

from __future__ import annotations

import csv
import json
import tempfile
import unittest
from pathlib import Path
from urllib.error import URLError

from github_issues_exporter import GitHubIssuesClient, IntegrationError, export_csv, parse_next_link


class FakeResponse:
    def __init__(self, payload: list[dict], link: str | None = None) -> None:
        self.payload = payload
        self.headers = {"Link": link} if link else {}

    def read(self) -> bytes:
        return json.dumps(self.payload).encode("utf-8")

    def __enter__(self) -> "FakeResponse":
        return self

    def __exit__(self, exc_type, exc, traceback) -> None:
        return None


class GitHubIntegrationTests(unittest.TestCase):
    def test_next_link_parser(self) -> None:
        raw = '<https://api.example.test/issues?page=2>; rel="next", <https://api.example.test/issues?page=4>; rel="last"'
        self.assertEqual(parse_next_link(raw), "https://api.example.test/issues?page=2")

    def test_pagination_validation_and_pull_request_filter(self) -> None:
        responses = iter(
            [
                FakeResponse(
                    [
                        {"id": 1, "number": 10, "title": " First  issue ", "state": "open", "html_url": "https://example/10"},
                        {"id": 2, "number": 11, "title": "PR", "state": "open", "html_url": "https://example/11", "pull_request": {}},
                    ],
                    '<https://api.example.test/issues?page=2>; rel="next"',
                ),
                FakeResponse(
                    [{"id": 3, "number": 12, "title": "Second issue", "state": "open", "html_url": "https://example/12"}]
                ),
            ]
        )

        client = GitHubIssuesClient(opener=lambda request, timeout: next(responses), sleeper=lambda _: None)
        issues = client.fetch_issues("owner", "repo", per_page=2, max_pages=3)
        self.assertEqual([issue["number"] for issue in issues], [10, 12])
        self.assertEqual(issues[0]["title"], "First issue")

    def test_retry_failure_is_clear(self) -> None:
        client = GitHubIssuesClient(
            opener=lambda request, timeout: (_ for _ in ()).throw(URLError("offline")),
            max_retries=1,
            sleeper=lambda _: None,
        )
        with self.assertRaisesRegex(IntegrationError, "after 2 attempts"):
            client.fetch_issues("owner", "repo")

    def test_csv_export(self) -> None:
        issues = [{"id": 1, "number": 2, "title": "A", "state": "open", "url": "https://example/2", "updated_at": ""}]
        with tempfile.TemporaryDirectory() as temp_dir:
            path = Path(temp_dir) / "issues.csv"
            export_csv(issues, path)
            with path.open(encoding="utf-8-sig") as handle:
                rows = list(csv.DictReader(handle))
            self.assertEqual(rows[0]["title"], "A")


if __name__ == "__main__":
    unittest.main()
