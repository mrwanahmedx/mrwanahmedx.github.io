# Demonstration A — Python Script Repair

## Problem

A small sales-report script stopped on real-world currency values such as `$1,250.00`. It also counted duplicate order exports and had no useful exception log.

## Reproduce the failure

```powershell
python reproduce_error.py
```

The original script raises a `ValueError` while converting a formatted currency string.

## Run the repaired workflow

```powershell
python repaired_sales_report.py sample_input/orders.csv -o output
```

## Evidence

```powershell
python -m unittest discover -s tests -v
```

The repaired implementation:

- normalizes currency safely with `Decimal`;
- ignores refunded rows;
- deduplicates repeated order IDs;
- records invalid amounts instead of crashing;
- writes a customer summary and exception log;
- includes regression tests for the failure that originally stopped the script.

## Business value

The client gets a repeatable report instead of a one-off patch, plus a clear list of rows that still need human review.

broken_sales_report.py

from __future__ import annotations

import csv
from collections import defaultdict
from pathlib import Path


def build_summary(input_csv: Path) -> dict[str, float]:
    totals: dict[str, float] = defaultdict(float)
    with input_csv.open(newline="", encoding="utf-8-sig") as handle:
        for row in csv.DictReader(handle):
            if row["status"].lower() == "refunded":
                continue
            totals[row["customer"]] += float(row["amount"])
    return dict(totals)

repaired_sales_report.py

from __future__ import annotations

import argparse
import csv
from collections import defaultdict
from decimal import Decimal, InvalidOperation
from pathlib import Path


ACCEPTED_STATUSES = {"paid", "completed"}


def parse_amount(raw: str) -> Decimal:
    cleaned = raw.strip().replace("$", "").replace(",", "")
    if not cleaned:
        raise ValueError("amount is empty")
    try:
        return Decimal(cleaned).quantize(Decimal("0.01"))
    except InvalidOperation as exc:
        raise ValueError(f"invalid amount: {raw!r}") from exc


def process_orders(input_csv: Path) -> tuple[list[dict[str, str]], list[dict[str, str]]]:
    totals: dict[str, Decimal] = defaultdict(lambda: Decimal("0.00"))
    seen_order_ids: set[str] = set()
    errors: list[dict[str, str]] = []

    with input_csv.open(newline="", encoding="utf-8-sig") as handle:
        reader = csv.DictReader(handle)
        required = {"order_id", "customer", "amount", "status"}
        missing = required.difference(reader.fieldnames or [])
        if missing:
            raise ValueError(f"missing columns: {', '.join(sorted(missing))}")

        for line_number, row in enumerate(reader, start=2):
            order_id = row["order_id"].strip()
            customer = " ".join(row["customer"].split())
            status = row["status"].strip().lower()

            if not order_id or not customer:
                errors.append({"line": str(line_number), "order_id": order_id, "issue": "missing order ID or customer"})
                continue
            if status not in ACCEPTED_STATUSES:
                continue
            if order_id in seen_order_ids:
                errors.append({"line": str(line_number), "order_id": order_id, "issue": "duplicate order ignored"})
                continue

            try:
                amount = parse_amount(row["amount"])
            except ValueError as exc:
                errors.append({"line": str(line_number), "order_id": order_id, "issue": str(exc)})
                continue

            seen_order_ids.add(order_id)
            totals[customer] += amount

    summary = [
        {"customer": customer, "paid_total": f"{amount:.2f}"}
        for customer, amount in sorted(totals.items())
    ]
    return summary, errors


def write_outputs(summary: list[dict[str, str]], errors: list[dict[str, str]], output_dir: Path) -> None:
    output_dir.mkdir(parents=True, exist_ok=True)
    with (output_dir / "customer_summary.csv").open("w", newline="", encoding="utf-8-sig") as handle:
        writer = csv.DictWriter(handle, fieldnames=["customer", "paid_total"])
        writer.writeheader()
        writer.writerows(summary)
    with (output_dir / "exceptions.csv").open("w", newline="", encoding="utf-8-sig") as handle:
        writer = csv.DictWriter(handle, fieldnames=["line", "order_id", "issue"])
        writer.writeheader()
        writer.writerows(errors)


def main() -> None:
    parser = argparse.ArgumentParser(description="Build a validated customer sales summary.")
    parser.add_argument("input_csv", type=Path)
    parser.add_argument("-o", "--output-dir", type=Path, default=Path("output"))
    args = parser.parse_args()
    summary, errors = process_orders(args.input_csv)
    write_outputs(summary, errors, args.output_dir)
    print(f"Customers summarized: {len(summary)}")
    print(f"Exceptions logged: {len(errors)}")


if __name__ == "__main__":
    main()

tests/test_sales_report.py

from __future__ import annotations

import csv
import tempfile
import unittest
from pathlib import Path

from repaired_sales_report import parse_amount, process_orders, write_outputs


ROOT = Path(__file__).resolve().parents[1]


class SalesReportTests(unittest.TestCase):
    def test_currency_normalization(self) -> None:
        self.assertEqual(str(parse_amount("$1,250.00")), "1250.00")

    def test_summary_deduplicates_and_logs_bad_rows(self) -> None:
        summary, errors = process_orders(ROOT / "sample_input" / "orders.csv")
        self.assertEqual(
            summary,
            [
                {"customer": "Acme Studio", "paid_total": "1250.00"},
                {"customer": "BrightCo", "paid_total": "800.50"},
                {"customer": "North & Pine", "paid_total": "175.00"},
            ],
        )
        self.assertEqual(len(errors), 2)
        self.assertIn("duplicate", errors[0]["issue"])
        self.assertIn("invalid amount", errors[1]["issue"])

    def test_outputs_are_written(self) -> None:
        summary, errors = process_orders(ROOT / "sample_input" / "orders.csv")
        with tempfile.TemporaryDirectory() as temp_dir:
            destination = Path(temp_dir)
            write_outputs(summary, errors, destination)
            with (destination / "customer_summary.csv").open(encoding="utf-8-sig") as handle:
                rows = list(csv.DictReader(handle))
            self.assertEqual(len(rows), 3)
            self.assertTrue((destination / "exceptions.csv").exists())


if __name__ == "__main__":
    unittest.main()
